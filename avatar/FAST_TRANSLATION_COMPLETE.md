## 🚀 Как использовать

### Просто и просто!

```tsx
'use client';

import { useLanguage } from '@/app/i18n';

export default function MyComponent() {
  const { t, language, setLanguage } = useLanguage();
  
  return (
    <div>
      <h1>{t('hero.title')}</h1>
      <p>Язык: {language}</p>
      <button onClick={() => setLanguage('en')}>English</button>
    </div>
  );
}
```
